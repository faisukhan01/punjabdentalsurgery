     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useState, type ReactNode } from "react";
     4→import { AnimatePresence, motion } from "framer-motion";
     5→import { format, parseISO } from "date-fns";
     6→import {
     7→  ArrowLeft,
     8→  ArrowRight,
     9→  CalendarCheck,
    10→  Check,
    11→  CheckCircle2,
    12→  Clock,
    13→  Loader2,
    14→  MoonStar,
    15→  SunMedium,
    16→  TriangleAlert,
    17→} from "lucide-react";
    18→import {
    19→  Dialog,
    20→  DialogContent,
    21→  DialogDescription,
    22→  DialogHeader,
    23→  DialogTitle,
    24→} from "@/components/ui/dialog";
    25→import { Button } from "@/components/ui/button";
    26→import { Input } from "@/components/ui/input";
    27→import { Label } from "@/components/ui/label";
    28→import { Textarea } from "@/components/ui/textarea";
    29→import { Skeleton } from "@/components/ui/skeleton";
    30→import { useToast } from "@/hooks/use-toast";
    31→import { useClinicStore } from "@/components/clinic/store";
    32→import {
    33→  SERVICE_NAMES,
    34→  TIME_SLOTS,
    35→  clinicTodayStr,
    36→} from "@/lib/clinic";
    37→
    38→/* ------------------------------- Types ------------------------------- */
    39→
    40→type Step = 1 | 2 | 3 | 4;
    41→
    42→interface Availability {
    43→  closed: boolean;
    44→  past: boolean;
    45→  taken: string[];
    46→}
    47→
    48→interface BookingForm {
    49→  name: string;
    50→  phone: string;
    51→  email: string;
    52→  message: string;
    53→}
    54→
    55→interface ConfirmedAppointment {
    56→  id: string;
    57→  name: string;
    58→  service: string;
    59→  date: string;
    60→  timeSlot: string;
    61→  status: string;
    62→}
    63→
    64→const STEP_LABELS = ["Service", "Date & Time", "Details", "Done"] as const;
    65→
    66→const MORNING_SLOTS = TIME_SLOTS.slice(0, 8) as readonly string[];
    67→const EVENING_SLOTS = TIME_SLOTS.slice(8) as readonly string[];
    68→
    69→const emptyForm: BookingForm = { name: "", phone: "", email: "", message: "" };
    70→
    71→/** Slide direction-aware step transitions (dir: 1 = forward, -1 = back). */
    72→const stepVariants = {
    73→  initial: (d: number) => ({ opacity: 0, x: 40 * d }),
    74→  animate: { opacity: 1, x: 0 },
    75→  exit: (d: number) => ({ opacity: 0, x: -40 * d }),
    76→};
    77→
    78→/* --------------------------- Small helpers --------------------------- */
    79→
    80→function prettyDate(dateStr: string): string {
    81→  try {
    82→    return format(parseISO(`${dateStr}T12:00:00`), "EEEE, d MMM yyyy");
    83→  } catch {
    84→    return dateStr;
    85→  }
    86→}
    87→
    88→function maxDateStr(): string {
    89→  const t = new Date();
    90→  return new Date(
    91→    t.getTime() + 5 * 60 * 60 * 1000 + 30 * 24 * 60 * 60 * 1000
    92→  )
    93→    .toISOString()
    94→    .slice(0, 10);
    95→}
    96→
    97→/* ------------------------------ Component ----------------------------- */
    98→
    99→export function BookingModal() {
   100→  const { toast } = useToast();
   101→  const bookingOpen = useClinicStore((s) => s.bookingOpen);
   102→  const preselectedService = useClinicStore((s) => s.preselectedService);
   103→  const closeBooking = useClinicStore((s) => s.closeBooking);
   104→
   105→  const [step, setStep] = useState<Step>(1);
   106→  const [dir, setDir] = useState<1 | -1>(1);
   107→
   108→  const [service, setService] = useState<string | null>(null);
   109→  const [date, setDate] = useState("");
   110→  const [slot, setSlot] = useState<string | null>(null);
   111→  const [availability, setAvailability] = useState<Availability | null>(null);
   112→  const [availLoading, setAvailLoading] = useState(false);
   113→
   114→  const [form, setForm] = useState<BookingForm>(emptyForm);
   115→  const [errors, setErrors] = useState<Partial<Record<keyof BookingForm, string>>>({});
   116→  const [submitError, setSubmitError] = useState<string | null>(null);
   117→  const [submitting, setSubmitting] = useState(false);
   118→  const [confirmed, setConfirmed] = useState<ConfirmedAppointment | null>(null);
   119→
   120→  const todayStr = useMemo(() => clinicTodayStr(), []);
   121→  const maxDate = useMemo(() => maxDateStr(), []);
   122→
   123→  /* ---------- Availability fetching ---------- */
   124→  const loadAvailability = useCallback(async (dateStr: string) => {
   125→    if (!dateStr) return;
   126→    setAvailLoading(true);
   127→    try {
   128→      const res = await fetch(
   129→        `/api/appointments/availability?date=${encodeURIComponent(dateStr)}`
   130→      );
   131→      const data = (await res.json()) as Partial<Availability>;
   132→      setAvailability({
   133→        closed: Boolean(data.closed),
   134→        past: Boolean(data.past),
   135→        taken: Array.isArray(data.taken) ? data.taken : [],
   136→      });
   137→    } catch {
   138→      // Optimistic: show all slots open if the check fails; server still guards.
   139→      setAvailability({ closed: false, past: false, taken: [] });
   140→    } finally {
   141→      setAvailLoading(false);
   142→    }
   143→  }, []);
   144→
   145→  /* ---------- Reset every time the dialog opens ---------- */
   146→  useEffect(() => {
   147→    if (!bookingOpen) return;
   148→    const pre =
   149→      preselectedService &&
   150→      (SERVICE_NAMES as readonly string[]).includes(preselectedService)
   151→        ? preselectedService
   152→        : null;
   153→    setDir(1);
   154→    setService(pre);
   155→    setStep(pre ? 2 : 1);
   156→    // Prefill today's date so patients see open slots immediately.
   157→    const today = clinicTodayStr();
   158→    setDate(today);
   159→    setSlot(null);
   160→    setAvailability(null);
   161→    void loadAvailability(today);
   162→    setForm(emptyForm);
   163→    setErrors({});
   164→    setSubmitError(null);
   165→    setConfirmed(null);
   166→  }, [bookingOpen, preselectedService, loadAvailability]);
   167→
   168→  const handleDateChange = (value: string) => {
   169→    setDate(value);
   170→    setSlot(null);
   171→    setAvailability(null);
   172→    if (value) void loadAvailability(value);
   173→  };
   174→
   175→  /* ---------- Navigation ---------- */
   176→  const goTo = (next: Step) => {
   177→    setDir(next > step ? 1 : -1);
   178→    setStep(next);
   179→  };
   180→
   181→  const back = () => {
   182→    if (step === 2) goTo(1);
   183→    if (step === 3) goTo(2);
   184→  };
   185→
   186→  /* ---------- Step 3 validation + submit ---------- */
   187→  const validateForm = (): boolean => {
   188→    const next: Partial<Record<keyof BookingForm, string>> = {};
   189→    if (form.name.trim().length < 2) next.name = "Please enter your full name.";
   190→    const digits = form.phone.replace(/\D/g, "");
   191→    if (digits.length < 10 || digits.length > 13)
   192→      next.phone = "Enter a valid phone number (e.g. 03XX-XXXXXXX).";
   193→    if (form.email.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(form.email.trim()))
   194→      next.email = "That email doesn't look right.";
   195→    setErrors(next);
   196→    return Object.keys(next).length === 0;
   197→  };
   198→
   199→  const submit = async () => {
   200→    if (!service || !date || !slot || !validateForm()) return;
   201→    setSubmitting(true);
   202→    setSubmitError(null);
   203→    try {
   204→      const res = await fetch("/api/appointments", {
   205→        method: "POST",
   206→        headers: { "Content-Type": "application/json" },
   207→        body: JSON.stringify({
   208→          name: form.name.trim(),
   209→          phone: form.phone.trim(),
   210→          email: form.email.trim() || undefined,
   211→          service,
   212→          date,
   213→          timeSlot: slot,
   214→          message: form.message.trim() || undefined,
   215→        }),
   216→      });
   217→
   218→      if (res.status === 409) {
   219→        setSlot(null);
   220→        void loadAvailability(date);
   221→        goTo(2);
   222→        toast({
   223→          variant: "destructive",
   224→          title: "That slot was just booked",
   225→          description: "Someone grabbed it before you — please pick another time.",
   226→        });
   227→        return;
   228→      }
   229→
   230→      const data = (await res.json().catch(() => null)) as
   231→        | { ok?: boolean; appointment?: ConfirmedAppointment; error?: string }
   232→        | null;
   233→
   234→      if (!res.ok || !data?.appointment) {
   235→        setSubmitError(
   236→          data?.error ??
   237→            "Something went wrong while booking. Please try again or call us directly."
   238→        );
   239→        return;
   240→      }
   241→
   242→      setConfirmed(data.appointment);
   243→      goTo(4);
   244→      toast({
   245→        title: "Booking confirmed 🎉",
   246→        description: `${data.appointment.service} — ${prettyDate(
   247→          data.appointment.date
   248→        )} at ${data.appointment.timeSlot}.`,
   249→      });
   250→    } catch {
   251→      setSubmitError("Network error — please check your connection and try again.");
   252→    } finally {
   253→      setSubmitting(false);
   254→    }
   255→  };
   256→
   257→  /* ---------- Render helpers ---------- */
   258→
   259→  const slotButtonClass = (s: string) => {
   260→    const taken = availability?.taken.includes(s) ?? false;
   261→    const selected = slot === s;
   262→    if (selected)
   263→      return "border-primary bg-primary text-primary-foreground shadow-[0_6px_18px_rgb(18,88,143,0.35)]";
   264→    if (taken) return "border-border/60 bg-muted text-muted-foreground/60 line-through";
   265→    return "border-border bg-card text-foreground hover:border-primary/50 hover:bg-secondary";
   266→  };
   267→
   268→  const renderSlotGroup = (label: string, icon: ReactNode, slots: readonly string[]) => (
   269→    <div>
   270→      <p className="mb-2 flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-[0.18em] text-muted-foreground">
   271→        {icon}
   272→        {label}
   273→      </p>
   274→      <div className="grid grid-cols-3 gap-2 sm:grid-cols-4">
   275→        {slots.map((s) => {
   276→          const taken = availability?.taken.includes(s) ?? false;
   277→          return (
   278→            <button
   279→              key={s}
   280→              type="button"
   281→              disabled={!availability || taken}
   282→              onClick={() => setSlot(s)}
   283→              aria-pressed={slot === s}
   284→              aria-label={`${s}${taken ? " (booked)" : ""}`}
   285→              className={`flex min-h-[44px] flex-col items-center justify-center rounded-xl border px-1 py-2 text-[13px] font-medium transition-all ${slotButtonClass(
   286→                s
   287→              )}`}
   288→            >
   289→              <span>{s}</span>
   290→              {taken && <span className="text-[10px] no-underline">Booked</span>}
   291→            </button>
   292→          );
   293→        })}
   294→      </div>
   295→    </div>
   296→  );
   297→
   298→  return (
   299→    <Dialog open={bookingOpen} onOpenChange={(open) => !open && closeBooking()}>
   300→      <DialogContent className="scrollbar-thin max-h-[92svh] w-[95vw] gap-0 overflow-y-auto rounded-3xl p-0 sm:max-w-lg">
   301→        <DialogHeader className="gap-1.5 border-b border-border/60 p-5 text-left sm:p-6">
   302→          <DialogTitle className="flex items-center gap-2.5 font-display text-xl">
   303→            <span className="flex size-9 items-center justify-center rounded-xl bg-primary text-primary-foreground">
   304→              <CalendarCheck className="size-4.5" aria-hidden />
   305→            </span>
   306→            Book Your Appointment
   307→          </DialogTitle>
   308→          <DialogDescription>
   309→            Under a minute — pick a service, choose a time, done.
   310→          </DialogDescription>
   311→
   312→          {/* Progress */}
   313→          <div className="mt-3">
   314→            <div className="h-1.5 w-full overflow-hidden rounded-full bg-muted">
   315→              <motion.div
   316→                className="h-full rounded-full bg-primary"
   317→                initial={false}
   318→                animate={{ width: `${(step / 4) * 100}%` }}
   319→                transition={{ duration: 0.4, ease: "easeOut" }}
   320→              />
   321→            </div>
   322→            <ol className="mt-2.5 flex items-center justify-between" aria-label="Booking steps">
   323→              {STEP_LABELS.map((label, i) => {
   324→                const n = (i + 1) as Step;
   325→                const active = n === step;
   326→                const done = n < step;
   327→                return (
   328→                  <li
   329→                    key={label}
   330→                    className={`flex items-center gap-1.5 text-[11px] font-semibold sm:text-xs ${
   331→                      active
   332→                        ? "text-primary"
   333→                        : done
   334→                          ? "text-primary"
   335→                          : "text-muted-foreground/70"
   336→                    }`}
   337→                    aria-current={active ? "step" : undefined}
   338→                  >
   339→                    {done ? (
   340→                      <Check className="size-3.5" aria-hidden />
   341→                    ) : (
   342→                      <span
   343→                        className={`size-1.5 rounded-full ${
   344→                          active ? "bg-primary" : "bg-muted-foreground/40"
   345→                        }`}
   346→                        aria-hidden
   347→                      />
   348→                    )}
   349→                    {label}
   350→                  </li>
   351→                );
   352→              })}
   353→            </ol>
   354→          </div>
   355→        </DialogHeader>
   356→
   357→        <div className="p-5 sm:p-6">
   358→          <AnimatePresence mode="wait" custom={dir} initial={false}>
   359→            {/* ------------------------------ STEP 1: SERVICE ------------------------------ */}
   360→            {step === 1 && (
   361→              <motion.div
   362→                key="step-1"
   363→                variants={stepVariants}
   364→                custom={dir}
   365→                initial="initial"
   366→                animate="animate"
   367→                exit="exit"
   368→                transition={{ duration: 0.3, ease: "easeOut" }}
   369→                className="flex flex-col gap-4"
   370→              >
   371→                <div>
   372→                  <h3 className="font-display text-lg font-semibold">What do you need help with?</h3>
   373→                  <p className="text-sm text-muted-foreground">Pick a service to continue.</p>
   374→                </div>
   375→                <div className="grid grid-cols-2 gap-2.5">
   376→                  {SERVICE_NAMES.map((name) => {
   377→                    const selected = service === name;
   378→                    return (
   379→                      <button
   380→                        key={name}
   381→                        type="button"
   382→                        onClick={() => setService(name)}
   383→                        aria-pressed={selected}
   384→                        className={`flex min-h-[44px] items-center gap-2 rounded-2xl border px-3.5 py-2.5 text-left text-[13px] font-medium leading-snug transition-all ${
   385→                          selected
   386→                            ? "border-primary bg-secondary text-primary shadow-[0_6px_18px_rgb(18,88,143,0.18)]"
   387→                            : "border-border bg-card text-foreground/85 hover:border-primary/40 hover:bg-secondary/50"
   388→                        }`}
   389→                      >
   390→                        <span
   391→                          className={`flex size-5 shrink-0 items-center justify-center rounded-full border ${
   392→                            selected ? "border-primary bg-primary text-primary-foreground" : "border-border"
   393→                          }`}
   394→                          aria-hidden
   395→                        >
   396→                          {selected && <Check className="size-3" />}
   397→                        </span>
   398→                        {name}
   399→                      </button>
   400→                    );
   401→                  })}
   402→                </div>
   403→                <Button
   404→                  className="mt-1 h-12 rounded-full text-[15px] font-semibold"
   405→                  disabled={!service}
   406→                  onClick={() => goTo(2)}
   407→                >
   408→                  Continue
   409→                  <ArrowRight className="size-4" aria-hidden />
   410→                </Button>
   411→              </motion.div>
   412→            )}
   413→
   414→            {/* --------------------------- STEP 2: DATE & TIME --------------------------- */}
   415→            {step === 2 && (
   416→              <motion.div
   417→                key="step-2"
   418→                variants={stepVariants}
   419→                custom={dir}
   420→                initial="initial"
   421→                animate="animate"
   422→                exit="exit"
   423→                transition={{ duration: 0.3, ease: "easeOut" }}
   424→                className="flex flex-col gap-4"
   425→              >
   426→                <div className="flex items-center justify-between gap-3">
   427→                  <div>
   428→                    <h3 className="font-display text-lg font-semibold">Choose date & time</h3>
   429→                    <p className="text-sm text-muted-foreground">
   430→                      {service && <span className="font-medium text-primary">{service}</span>}
   431→                    </p>
   432→                  </div>
   433→                  <Button variant="ghost" size="sm" className="h-9 shrink-0" onClick={back}>
   434→                    <ArrowLeft className="size-4" aria-hidden />
   435→                    Back
   436→                  </Button>
   437→                </div>
   438→
   439→                <div className="flex flex-col gap-1.5">
   440→                  <Label htmlFor="booking-date">Appointment date</Label>
   441→                  <Input
   442→                    id="booking-date"
   443→                    type="date"
   444→                    value={date}
   445→                    min={todayStr}
   446→                    max={maxDate}
   447→                    onChange={(e) => handleDateChange(e.target.value)}
   448→                    className="h-11 rounded-xl"
   449→                  />
   450→                </div>
   451→
   452→                {date && availability?.closed && (
   453→                  <div className="flex items-start gap-2.5 rounded-2xl border border-amber-300/70 bg-amber-50 p-3.5 text-sm text-amber-900">
   454→                    <TriangleAlert className="mt-0.5 size-4 shrink-0 text-amber-500" aria-hidden />
   455→                    The clinic is closed on that day — please pick another day.
   456→                  </div>
   457→                )}
   458→                {date && availability?.past && (
   459→                  <div className="flex items-start gap-2.5 rounded-2xl border border-amber-300/70 bg-amber-50 p-3.5 text-sm text-amber-900">
   460→                    <TriangleAlert className="mt-0.5 size-4 shrink-0 text-amber-500" aria-hidden />
   461→                    That date has already passed — please pick today or a future date.
   462→                  </div>
   463→                )}
   464→
   465→                {date && availLoading && (
   466→                  <div className="flex flex-col gap-3">
   467→                    <Skeleton className="h-3.5 w-20" />
   468→                    <div className="grid grid-cols-3 gap-2 sm:grid-cols-4">
   469→                      {Array.from({ length: 8 }).map((_, i) => (
   470→                        <Skeleton key={i} className="h-[44px] rounded-xl" />
   471→                      ))}
   472→                    </div>
   473→                  </div>
   474→                )}
   475→
   476→                {date && availability && !availability.closed && !availability.past && !availLoading && (
   477→                  <div className="flex flex-col gap-4">
   478→                    {availability.taken.length >= TIME_SLOTS.length ? (
   479→                      <div className="flex items-start gap-2.5 rounded-2xl border border-amber-300/70 bg-amber-50 p-3.5 text-sm text-amber-900">
   480→                        <TriangleAlert className="mt-0.5 size-4 shrink-0 text-amber-500" aria-hidden />
   481→                        Every slot for {prettyDate(date)} is booked. Please try another day.
   482→                      </div>
   483→                    ) : (
   484→                      <>
   485→                        {renderSlotGroup("Morning", <SunMedium className="size-3.5" />, MORNING_SLOTS)}
   486→                        {renderSlotGroup("Evening", <MoonStar className="size-3.5" />, EVENING_SLOTS)}
   487→                      </>
   488→                    )}
   489→                  </div>
   490→                )}
   491→
   492→                {date && !availability && !availLoading && (
   493→                  <p className="text-sm text-muted-foreground">
   494→                    Select a date to see available times.
   495→                  </p>
   496→                )}
   497→
   498→                <Button
   499→                  className="mt-1 h-12 rounded-full text-[15px] font-semibold"
   500→                  disabled={!date || !slot || Boolean(availability?.closed)}
   501→                  onClick={() => goTo(3)}
   502→                >
   503→                  Continue
   504→                  <ArrowRight className="size-4" aria-hidden />
   505→                </Button>
   506→              </motion.div>
   507→            )}
   508→
   509→            {/* ----------------------------- STEP 3: DETAILS ----------------------------- */}
   510→            {step === 3 && (
   511→              <motion.div
   512→                key="step-3"
   513→                variants={stepVariants}
   514→                custom={dir}
   515→                initial="initial"
   516→                animate="animate"
   517→                exit="exit"
   518→                transition={{ duration: 0.3, ease: "easeOut" }}
   519→                className="flex flex-col gap-4"
   520→              >
   521→                <div className="flex items-center justify-between gap-3">
   522→                  <div>
   523→                    <h3 className="font-display text-lg font-semibold">Your details</h3>
   524→                    <p className="text-sm text-muted-foreground">We&apos;ll call to confirm.</p>
   525→                  </div>
   526→                  <Button variant="ghost" size="sm" className="h-9 shrink-0" onClick={back}>
   527→                    <ArrowLeft className="size-4" aria-hidden />
   528→                    Back
   529→                  </Button>
   530→                </div>
   531→
   532→                {/* Summary line */}
   533→                <div className="flex flex-wrap items-center gap-x-4 gap-y-1 rounded-2xl bg-secondary/70 px-4 py-3 text-[13px] text-secondary-foreground">
   534→                  <span className="font-semibold">{service}</span>
   535→                  <span className="inline-flex items-center gap-1">
   536→                    <CalendarCheck className="size-3.5" aria-hidden />
   537→                    {prettyDate(date)}
   538→                  </span>
   539→                  <span className="inline-flex items-center gap-1">
   540→                    <Clock className="size-3.5" aria-hidden />
   541→                    {slot}
   542→                  </span>
   543→                </div>
   544→
   545→                <div className="flex flex-col gap-3.5">
   546→                  <div className="flex flex-col gap-1.5">
   547→                    <Label htmlFor="booking-name">Full name *</Label>
   548→                    <Input
   549→                      id="booking-name"
   550→                      value={form.name}
   551→                      onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   552→                      placeholder="e.g. Ali Raza"
   553→                      autoComplete="name"
   554→                      aria-invalid={Boolean(errors.name)}
   555→                      className="h-11 rounded-xl"
   556→                    />
   557→                    {errors.name && <p className="text-xs text-destructive">{errors.name}</p>}
   558→                  </div>
   559→
   560→                  <div className="flex flex-col gap-1.5">
   561→                    <Label htmlFor="booking-phone">Phone number *</Label>
   562→                    <Input
   563→                      id="booking-phone"
   564→                      type="tel"
   565→                      inputMode="tel"
   566→                      value={form.phone}
   567→                      onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value }))}
   568→                      placeholder="03XX-XXXXXXX"
   569→                      autoComplete="tel"
